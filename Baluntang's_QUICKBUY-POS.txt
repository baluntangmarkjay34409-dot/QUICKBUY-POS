print("=================================")
print("           QUICKBUY POS")
print("     MJVB-Simple Store System")
print("=================================")
while True:
    print("\n=================================")
    print("          ACCOUNT MENU")
    print("=================================")
    print("1. Sign Up")
    print("2. Log In")
    print("3. Exit")
    account_choice = input("Enter your choice: ")
    if account_choice == "1":
        print("\n--- SIGN UP ---")
        username = input("Create username: ")
        password = input("Create password: ")
        print("\nAccount created successfully!")
        print("You can now log in to QuickBuy.")
    elif account_choice == "2":
        if username == "" or password == "":
            print("\nNo account found!")
            print("Please Sign Up first.")
            continue
        print("\n--- LOGIN ---")
        login_username = input("Username: ")
        login_password = input("Password: ")
        if login_username == username and login_password == password:
            print("\nLogin successful!")
            print("Welcome to QuickBuy,", username)
            while True:
                print("\n=================================")
                print("          QUICKBUY MENU")
                print("=================================")
                print("1. View Products")
                print("2. Buy Product")
                print("3. Logout")
                choice = input("Enter your choice: ")
                if choice == "1":
                    print("\n--- PRODUCTS ---")
                    print("1. Bread       - ₱25")
                    print("2. Soft Drink  - ₱30")
                    print("3. Chips       - ₱20")
                elif choice == "2":
                    print("\n--- BUY PRODUCT ---")
                    print("1. Bread       - ₱25")
                    print("2. Soft Drink  - ₱30")
                    print("3. Chips       - ₱20")
                    product = input("Choose product (1-3): ")
                    try:
                        quantity = int(input("Enter quantity: "))
                        if quantity <= 0:
                            print("Invalid quantity!")
                            continue
                    except ValueError:
                        print("Please enter a valid number!")
                        continue
                    if product == "1":
                        item = "Bread"
                        price = 25
                    elif product == "2":
                        item = "Soft Drink"
                        price = 30
                    elif product == "3":
                        item = "Chips"
                        price = 20
                    else:
                        print("Invalid product!")
                        continue
                    total = price * quantity

                    print("\n--- ORDER SUMMARY ---")
                    print("Product:", item)
                    print("Quantity:", quantity)
                    print("Total: ₱", total)
                    try:
                        payment = float(input("Enter payment: ₱"))
                        if payment >= total:
                            change = payment - total
                            print("Change: ₱", change)
                            print("Thank you for shopping at QuickBuy!")
                        else:
                            print("Insufficient payment!")
                    except ValueError:
                        print("Please enter a valid payment amount!")
                elif choice == "3":
                    print("\nYou have been logged out.")
                    print("Goodbye,", username)
                    break
                else:
                    print("Invalid choice. Please try again.")
        else:
            print("\nLogin failed!")
            print("Incorrect username or password.")
    elif account_choice == "3":
        print("\nThank you for using QuickBuy POS!")
        print("Goodbye!")
        break
    else:
        print("\nInvalid choice. Please try again.")
